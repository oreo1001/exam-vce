-- 학습 기록
create table study_answers (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users(id) on delete cascade not null,
  subject text not null,
  question_num integer not null,
  selected text[] not null,
  correct boolean not null,
  updated_at timestamptz default now(),
  unique(user_id, subject, question_num)
);

-- 오답노트
create table wrong_questions (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users(id) on delete cascade not null,
  subject text not null,
  question_num integer not null,
  added_at timestamptz default now(),
  unique(user_id, subject, question_num)
);

-- 시험 결과
create table test_results (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users(id) on delete cascade not null,
  subject text not null,
  score integer not null,
  total integer not null,
  elapsed integer not null,
  taken_at timestamptz default now()
);

-- 본인 데이터만 접근 가능하도록 RLS 설정
alter table study_answers enable row level security;
alter table wrong_questions enable row level security;
alter table test_results enable row level security;

create policy "본인 데이터만" on study_answers for all using (auth.uid() = user_id);
create policy "본인 데이터만" on wrong_questions for all using (auth.uid() = user_id);
create policy "본인 데이터만" on test_results for all using (auth.uid() = user_id);

-- ── Keep-alive (Free plan 자동 pause 방지) ─────────────────────────
-- 빈 SELECT 만으로는 "활동"으로 안 잡히는 경우가 있어, 매일 실제 UPDATE 를 한 번 일으킨다.
-- GitHub Actions(.github/workflows/supabase-keepalive.yml)가 anon 키로 rpc/keepalive 를 호출한다.
create table if not exists keepalive (
  id int primary key default 1 check (id = 1),
  pinged_at timestamptz not null default now(),
  count bigint not null default 0
);
alter table keepalive enable row level security; -- 정책 없음 → 테이블 직접 접근 불가

create or replace function public.keepalive()
returns timestamptz
language sql
security definer
set search_path = public
as $$
  insert into keepalive (id, pinged_at, count) values (1, now(), 1)
  on conflict (id) do update set pinged_at = now(), count = keepalive.count + 1
  returning pinged_at;
$$;

revoke all on function public.keepalive() from public;
grant execute on function public.keepalive() to anon;
